cmake_minimum_required(VERSION 3.20)
project(kernel_event_order LANGUAGES C)
if(NOT WIN32)
    message(FATAL_ERROR "This regression harness currently targets Windows.")
endif()
add_subdirectory(../.. runtime)
enable_testing()
foreach(name dpc_atomic dpc_timeout dpc_blocking)
    add_executable(${name} ${name}.c)
    target_include_directories(${name} PRIVATE ../../src/kernel)
    target_link_libraries(${name} PRIVATE xboxrecomp)
    add_test(NAME ${name} COMMAND ${name})
endforeach()
